const {chromium} = require('/Users/bytedance/.cache/codex-runtimes/codex-primary-runtime/dependencies/node/node_modules/playwright');
const fs = require('node:fs');
(async () => {
 const browser = await chromium.launch({channel:'msedge',headless:true});
 try {
 const base = process.argv[2] || 'http://127.0.0.1:8765';
 const label = process.argv[3] || 'local';
 const results = [];
 for (const width of [1440, 960, 769, 390, 320]) {
  const context = await browser.newContext({viewport:{width,height:900},permissions:['clipboard-read','clipboard-write']});
  const page = await context.newPage();
  const errors = [];
  page.on('pageerror', e => errors.push(e.message));
  for (const path of ['/', '/blog/', '/blog/2026-10-v2.4-release.html']) {
   const response = await page.goto(base + path, {waitUntil:'networkidle'});
   if (!response.ok()) throw Error(`${path} HTTP ${response.status()}`);
   const dimensions = await page.evaluate(() => ({scroll:document.documentElement.scrollWidth,viewport:innerWidth}));
   if (dimensions.scroll > width) throw Error(`${path} overflows at ${width}: ${JSON.stringify(dimensions)}`);
   if (path === '/') {
    await page.locator('.hero-badge').filter({hasText:'v2.4.0'}).waitFor();
    const command = await page.locator('.hero-install code').innerText();
    if (!command.includes('/releases/download/v2.4.0/nexa_lang-2.4.0-py3-none-any.whl')) throw Error('Wrong install URL');
    await page.getByRole('button',{name:'Copy install command'}).click();
    const copied = await page.evaluate(()=>navigator.clipboard.readText());
    if (copied !== command) throw Error('Clipboard mismatch');
    if (width < 768) {
     await page.getByRole('button',{name:'Toggle navigation'}).click();
     await page.locator('.nav-links').getByRole('link',{name:'Blog',exact:true}).click();
     if (!page.url().endsWith('/blog/')) throw Error('Mobile navigation failed');
     await page.goto(base + '/', {waitUntil:'networkidle'});
    }
   }
   if (path === '/blog/') {
    await page.locator('.blog-sort').selectOption('oldest');
    if (await page.locator('.blog-card-date').first().innerText() !== '2026.01.15') throw Error('Oldest-first sort failed');
    await page.locator('.blog-sort').selectOption('newest');
    if (await page.locator('.blog-card-date').first().innerText() !== '2026.10.23') throw Error('Newest-first sort failed');
    await page.getByPlaceholder('Search posts...').fill('2.4.0');
    if (await page.locator('.blog-card:visible').count() !== 1) throw Error('Release search failed');
    await page.getByPlaceholder('Search posts...').fill('');
   }
   const name = path === '/' ? 'home' : path === '/blog/' ? 'blog' : 'release';
   await page.screenshot({path:`/private/tmp/nexa-${label}-${name}-${width}.png`});
   results.push({path,width,status:response.status(),...dimensions});
  }
  if (errors.length) throw Error(`Browser errors: ${JSON.stringify(errors)}`);
  await context.close();
 }
 fs.writeFileSync(`/private/tmp/nexa-${label}-browser.json`,JSON.stringify(results,null,2)+'\n');
 console.log(JSON.stringify(results));
 } finally { await browser.close(); }
})().catch(e=>{console.error(e);process.exitCode=1;});
